#!/usr/bin/env python3
"""Actual packaged V1.5 runtime vs inherited V1.4 coverage renderer.
Full source-sized canvases are compared. This is not a garment reapproval test.
"""
from pathlib import Path
import base64, importlib.util, json, shutil, time, sys
from playwright.sync_api import sync_playwright
R=Path(__file__).resolve().parents[1];E=R/'evidence/edge_v1_5';E.mkdir(exist_ok=True)
spec=importlib.util.spec_from_file_location('build',R/'tools/build.py');build=importlib.util.module_from_spec(spec);spec.loader.exec_module(build)
checks=[];rows=[];controls=[];errors=[];requests=[];failure=None;t0=time.monotonic()
def check(name,ok,detail=None):
 checks.append({'name':name,'pass':bool(ok),'detail':detail});print(('PASS ' if ok else 'FAIL ')+name,flush=True)
 if not ok:raise AssertionError(name+': '+str(detail))
def wait(p,expr,seconds=45):
 end=time.monotonic()+seconds
 while time.monotonic()<end:
  if p.evaluate('()=>Boolean('+expr+')'):return
  time.sleep(.04)
 raise TimeoutError(expr)
def save_canvas(p,expr,name):
 raw=p.evaluate('()=>'+expr+".toDataURL('image/png').split(',')[1]")
 (E/name).write_bytes(base64.b64decode(raw))
with sync_playwright() as pw:
 b=pw.chromium.launch(headless=True,executable_path=shutil.which('chromium'),args=['--no-sandbox','--disable-dev-shm-usage'])
 p=b.new_page(viewport={'width':390,'height':844},device_scale_factor=3)
 p.on('pageerror',lambda e:errors.append(str(e)));p.on('request',lambda r:requests.append(r.url) if r.url.startswith(('http:','https:')) else None)
 try:
  html=(R/'app.template.html').read_text().replace('<!--STYLE-->','<style>'+(R/'src/application.css').read_text()+'</style>').replace('<!--SCRIPTS-->','')
  p.set_content(html,wait_until='load');p.evaluate('globalThis.HEWRS_EMBEDDED_IMAGES={}');batch={};n=0
  for f in sorted([*(R/'assets').glob('*.png'),*(R/'assemblies').glob('*.png')]):
   v=base64.b64encode(f.read_bytes()).decode();batch[f.stem]=v;n+=len(v)
   if n>4_000_000:p.evaluate('(v)=>Object.assign(HEWRS_EMBEDDED_IMAGES,v)&&true',batch);batch={};n=0
  if batch:p.evaluate('(v)=>Object.assign(HEWRS_EMBEDDED_IMAGES,v)&&true',batch)
  for f in build.SCRIPTS:p.add_script_tag(content=(R/f).read_text())
  wait(p,'globalThis.HEWRS_READY===true')
  check('V1.5 starts',p.evaluate("HEWRSApp.version==='HEWRS_CONNECTED_APP_V1_5'"))
  # Test images in small ROI support only, test-only globals.
  for mode in ['no_tie','tied']:
   for tag in ['PERMITTED','PROTECTED']:
    import numpy as np
    from PIL import Image
    a=np.array(Image.open(E/f'{mode}_{tag}_MASK_ROI.png'))
    if a.ndim==3:a=a[:,:,3]
    p.evaluate('(o)=>{globalThis[o.k]=new Uint8Array(o.v)}',{'k':mode+'_'+tag,'v':a.reshape(-1).tolist()})
  p.evaluate('''()=>{
   globalThis.W=996;globalThis.H=2748;
   globalThis.pix=c=>c.getContext('2d',{willReadFrequently:true}).getImageData(0,0,W,H).data;
   globalThis.same=(a,b)=>{const aa=new Uint32Array(a.buffer,a.byteOffset,a.length/4),bb=new Uint32Array(b.buffer,b.byteOffset,b.length/4);return aa.length===bb.length&&aa.every((v,i)=>v===bb[i]);};
   globalThis.reference=document.createElement('canvas');
   globalThis.oldRenderer=HEWRSAtomicRenderer.create(reference,HEWRSApp.connection,{resolveUrl:HEWRSApp.resolveUrl,rendererFactory:HEWRSDS035CoverageRenderer.create});
   globalThis.imcache=new Map();
   globalThis.get=(d)=>{if(imcache.has(d.sha256))return imcache.get(d.sha256);const q=new Promise((ok,bad)=>{const im=new Image();im.onload=()=>{const c=document.createElement('canvas');c.width=W;c.height=H;c.getContext('2d').drawImage(im,0,0);const a=pix(c);c.width=c.height=1;ok(a);};im.onerror=bad;im.src=HEWRSApp.resolveUrl(d);});imcache.set(d.sha256,q);if(imcache.size>8)imcache.delete(imcache.keys().next().value);return q;};
   globalThis.audit=async(s)=>{
    const a=HEWRSApp,mode=s.state==='NO_TIE'?'no_tie':'tied';
    await oldRenderer.render(s);const before=pix(reference);
    const old=pix(document.getElementById('avatar')).slice(),pending=a.apply(s,{save:false}),held=same(old,pix(document.getElementById('avatar')));await pending;
    const after=pix(document.getElementById('avatar')),aa=new Uint32Array(after.buffer),bb=new Uint32Array(before.buffer);
    const permitted=globalThis[mode+'_PERMITTED'],protectedMask=globalThis[mode+'_PROTECTED'];
    const mt=a.connection.assemblies.forSelection(s.suitId,s.shoeId,s.shirtId),tie=s.state==='NO_TIE'?null:await get(mt.ties[s.state].display_layer),jacket=await get(mt.static.jacket),jkeep=await get(mt.static.jacket.display_alpha_mask);
    let changed=0,outside=0,protectedChanged=0,tieChanged=0,jacketChanged=0,alphaChanged=0,headChanged=0,residualEmptyPixels=0;
    for(let y=350;y<950;y++)for(let x=340;x<660;x++){const ri=(y-350)*320+x-340,i=y*W+x;if(permitted[ri]&&(!tie||!tie[4*i+3])&&!after[4*i+3])residualEmptyPixels++;}
    for(let i=0;i<aa.length;i++)if(aa[i]!==bb[i]){
     changed++;const x=i%W,y=Math.floor(i/W),within=x>=340&&x<660&&y>=350&&y<950,ri=(y-350)*320+x-340;
     if(!within||!permitted[ri])outside++;if(within&&protectedMask[ri])protectedChanged++;
     if(tie&&tie[4*i+3])tieChanged++;
     if(jacket[i*4+3]===255&&jkeep[i*4+3]===255)jacketChanged++;
     if(after[i*4+3]!==before[i*4+3])alphaChanged++;
     if(y<430||y>=950)headChanged++;
    }
    const probes=[[567,540],[562,560],[595,470]].map(([x,y])=>({xy:[x,y],before:Array.from(before.slice((y*W+x)*4,(y*W+x)*4+4)),after:Array.from(after.slice((y*W+x)*4,(y*W+x)*4+4))}));
    return {...s,changed,outside,protectedChanged,tieChanged,jacketChanged,alphaChanged,headChanged,residualEmptyPixels,held,identity:JSON.stringify(s)===JSON.stringify(a.state().selection),probes};
   };
  }''')
  wide=p.evaluate('HEWRSApp.connection.assemblies.widerDS035SuitIds')
  modes=p.evaluate("['NO_TIE',...Object.keys(HEWRSApp.connection.manifest.ties)]")
  # Every new edge route, not a replay of all approved wardrobe renders.
  for sid in wide:
   for i in range(0,len(modes),8):
    got=p.evaluate('''async q=>{const out=[];for(const state of q.modes)out.push(await audit({suitId:q.sid,shirtId:'DS035',state,shoeId:'shoe-8',watchId:null}));return out;}''',{'sid':sid,'modes':modes[i:i+8]});rows.extend(got)
    if any(any(r[k] for k in ['outside','protectedChanged','tieChanged','jacketChanged','headChanged','residualEmptyPixels']) or not r['held'] or not r['identity'] for r in got):raise AssertionError('Protected output changed '+sid+': '+json.dumps(got))
   check(sid+' all 48 modes preserve protected pixels and close edge holes',True,{'modes':48})
   (E/'BROWSER_PROGRESS.json').write_text(json.dumps({'rows':len(rows),'suit':sid}))
  # Original-template routes should be completely untouched.
  for sid in ['S05','S01','S04','S16']:
   states=modes+(['REFERENCE'] if sid=='S05' else [])
   for i in range(0,len(states),8):
    controls.extend(p.evaluate('''async q=>{const out=[];for(const state of q.modes){const s={suitId:q.sid,shirtId:'DS035',state,shoeId:'shoe-8',watchId:null};await oldRenderer.render(s);await HEWRSApp.apply(s,{save:false});out.push({...s,identical:same(pix(reference),pix(document.getElementById('avatar')))});}return out;}''',{'sid':sid,'modes':states[i:i+8]}))
  check('193 original-template DS035 outputs remain identical',len(controls)==193 and all(r['identical'] for r in controls))
  others=p.evaluate('''async()=>{const out=[];for(const [shirtId,state] of [['DS017','T017'],['DS036','NO_TIE'],['DS051','T047'],['DS027','NO_TIE'],['DS041','NO_TIE']]){const s={suitId:'S05',shirtId,state,shoeId:'shoe-8',watchId:null};await oldRenderer.render(s);await HEWRSApp.apply(s,{save:false});out.push({...s,identical:same(pix(reference),pix(document.getElementById('avatar')))});}return out;}''')
  check('Other-shirt controls unchanged',all(x['identical'] for x in others),others)
  check('The two reported body-edge probes and collar fringe are corrected',all(max(r['probes'][0]['after'][:3])>180 and max(r['probes'][1]['after'][:3])>180 for r in rows if r['state']=='T017'))
  # Genuine captures of app output and old app output, same native coordinates.
  for mode,state in [('tied','T017'),('no_tie','NO_TIE')]:
   p.evaluate('async state=>await audit({suitId:"S11",shirtId:"DS035",state,shoeId:"shoe-8",watchId:null})',state)
   save_canvas(p,'reference',mode+'_BEFORE_NATIVE.png');save_canvas(p,'document.getElementById("avatar")',mode+'_AFTER_NATIVE.png')
  # Actual Engine path and display, then a phone/full-detail screenshot.
  res=p.evaluate('''async()=>{HEWRSApp.showPage('home');HEWRSApp.setMode('engine');const r=await HEWRSApp.generate({suitId:'S11',shirt:'DS035',tie:'T017',shoeId:'shoe-8',watchId:null,localDate:'2026-09-22',occasion:'clinic',formality:'any',style:'AUTO'});return {options:r.options.length,selection:HEWRSApp.state().selection,events:HEWRSApp.store.snapshot().events.length};}''')
  check('Engine displays exact cleaned DS035 selection without logging wear',res['options']>0 and res['selection']['suitId']=='S11' and res['selection']['state']=='T017' and res['events']==0,res)
  p.screenshot(path=str(E/'APP_PHONE_390x844_DPR3.png'))
  # Use actual collar control selector from the app.
  print('DETAIL BUTTONS',p.locator('button').evaluate_all('(a)=>a.filter(b=>/collar|detail/i.test(b.textContent)).map(b=>({id:b.id,text:b.textContent}))'),flush=True)
  ids=p.locator('button').evaluate_all('(a)=>a.filter(b=>/collar/i.test(b.textContent)).map(b=>b.id)')
  if ids:
   p.locator('#'+ids[0]).click();p.screenshot(path=str(E/'APP_COLLAR_390x844_DPR3.png'));p.locator('#'+ids[0]).click()
  rapid=p.evaluate('''async()=>{const a=HEWRSApp;const before=pix(document.getElementById('avatar')).slice(),work=[];for(const suitId of ['S02','S11','S18'])work.push(a.apply({suitId,shirtId:'DS035',state:'T022',shoeId:'shoe-8',watchId:null},{save:false}));const held=same(before,pix(document.getElementById('avatar')));await Promise.all(work);return {held,last:a.state().selection.suitId};}''')
  check('Rapid switching retains complete frames and final identity',rapid['held'] and rapid['last']=='S18',rapid)
  inv=p.evaluate('''async()=>{const before=pix(document.getElementById('avatar')).slice(),s=HEWRSApp.state().selection;let rejected=false;try{await HEWRSApp.apply({...s,shirtId:'DS036'},{save:false});}catch{rejected=true;}return {rejected,held:same(before,pix(document.getElementById('avatar'))),identity:JSON.stringify(s)===JSON.stringify(HEWRSApp.state().selection)};}''')
  check('Other-shirt request still rejects without a replacement',all(inv.values()),inv)
  check('No JavaScript errors',not errors,errors);check('No external network requests',not requests,requests)
 except Exception as e:
  failure=repr(e);print('FAILURE',failure,flush=True)
  try:p.screenshot(path=str(E/'FAILED_STATE.png'))
  except Exception:pass
 finally:
  result={'schema':'hewrs.ds035.edge.browser.v1_5','checks':checks,'checks_passed':sum(x['pass'] for x in checks),'checks_failed':sum(not x['pass'] for x in checks),'failure':failure,'edge_comparison_count':len(rows),'edge_comparisons':rows,'original_route_control_count':len(controls),'original_route_controls':controls,'errors':errors,'external_requests':requests,'browser':b.version,'seconds':round(time.monotonic()-t0,2),'transport':'Exact packaged source and images supplied in memory to Chromium; compared with inherited V1.4 coverage compositor','not_tested':['Physical iPhone','Safari','Hosted delivery','Persistent browser restart']}
  (E/'BROWSER.json').write_text(json.dumps(result,indent=2)+'\n');b.close()
print(json.dumps({k:result[k] for k in ['checks_passed','checks_failed','failure','edge_comparison_count','original_route_control_count','seconds']}),flush=True)
if failure or result['checks_failed']:sys.exit(1)
