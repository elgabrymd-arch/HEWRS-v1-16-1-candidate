'use strict';
const fs=require('fs'),path=require('path'),vm=require('vm'),assert=require('assert/strict'),crypto=require('crypto');
const R=path.resolve(__dirname,'..'),p=JSON.parse(fs.readFileSync(R+'/data/inputs.json'));
const sandbox={console,structuredClone,Date,Math,JSON,setTimeout,clearTimeout,Uint8ClampedArray};
const cx=vm.createContext(sandbox);
for(const file of ['vendor/hewrs-logic.js','vendor/approved-scores.js','vendor/ensemble-completion.js','vendor/release-guards.js','vendor/request-contract.js','src/controller.js','vendor/watch-classification.js','src/canonical-bindings-browser.js','data/approved-suits.js','src/suit-resolver97-browser.js','src/approved-suit-sources.js','data/suit-assembly.js','data/ds035-coverage.js','src/suit-assemblies.js','src/connection.js','src/local-state.js'])vm.runInContext(fs.readFileSync(R+'/'+file,'utf8'),cx,{filename:file});

const before=JSON.stringify(p),app=cx.HEWRSCleanConnection.create(p),checks=[];
const equal=(a,b)=>assert.equal(JSON.stringify(a),JSON.stringify(b));
function check(name,fn){try{fn();checks.push({name,pass:true});}catch(e){checks.push({name,pass:false,error:e.stack});}}
const s={suitId:'S11',shirtId:'DS035',state:'T017',shoeId:'shoe-8',watchId:null};
check('S05 original descriptor tree and masks unchanged',()=>equal(app.assemblies.forSelection('S05','shoe-8','DS035'),p.manifest));
check('Four original connected suit IDs unchanged',()=>equal(app.assemblies.dynamicSuitIds,['S01','S04','S05','S16']));
check('Fourteen wider suits use explicit registered identities',()=>{assert.equal(app.assemblies.widerDS035SuitIds.length,14);for(const id of app.assemblies.widerDS035SuitIds)assert.equal(app.suitSources.resolveCanonical(id).template,'S11');});
let routed=0;
check('All 672 DS035 wider-suit requests validate without substitute shirt or tie',()=>{for(const suitId of app.assemblies.widerDS035SuitIds)for(const state of ['NO_TIE',...Object.keys(p.manifest.ties)]){const q={...s,suitId,state};equal(app.validateSelection(q),q);equal(app.bridge.stateRequest(suitId,'DS035',state),{suitId,shirtId:'DS035',state});routed++;}assert.equal(routed,672);});
check('All wider-suit manifests preserve source shirt components and selected paired suit',()=>{for(const id of app.assemblies.widerDS035SuitIds){const m=app.assemblies.forSelection(id,'shoe-8','DS035'),pair=app.suitSources.resolveCanonical(id);equal(m.shirts,p.manifest.shirts);equal(m.ties,p.manifest.ties);assert.equal(m.static.jacket.sha256,pair.jacket.sha256);assert.equal(m.static.trousers.sha256,pair.trousers.sha256);assert.equal(m.ds035_s11_coverage.selected_suit_id,id);}});
let rejected=0;
check('Other shirts stay outside authorized wider-suit routes',()=>{for(const suitId of app.assemblies.widerDS035SuitIds)for(const shirtId of p.manifest.shirt_order.filter(x=>x!=='DS035')){assert.throws(()=>app.validateSelection({...s,suitId,shirtId}));rejected++;}assert.equal(rejected,686);});
check('REFERENCE remains S05 only, never an alternate tied selection',()=>{for(const suitId of app.assemblies.widerDS035SuitIds)assert.throws(()=>app.validateSelection({...s,suitId,state:'REFERENCE'}));app.validateSelection({...s,suitId:'S05',state:'REFERENCE'});});
check('Unknown tie ID and retired shirt reject',()=>{assert.throws(()=>app.validateSelection({...s,state:'T048'}));assert.throws(()=>app.validateSelection({...s,shirtId:'DS049'}));});
check('Unknown suit and registered-shoe mismatch reject',()=>{assert.throws(()=>app.validateSelection({...s,suitId:'S19'}));assert.throws(()=>app.validateSelection({...s,shoeId:'shoe-50'}));});
check('Engine request scope unchanged',()=>{for(const suitId of app.assemblies.widerDS035SuitIds)assert.throws(()=>app.makeRequest({suitId,shirt:'DS035',tie:'T017',shoeId:'shoe-8',localDate:'2026-09-22'}));});
check('Original component hash mappings do not change',()=>{for(const [h,rel]of Object.entries(p.assets))assert.equal(app.assetPaths[h],rel);});
check('Exactly two coverage variants are bound separately',()=>{const data=cx.HEWRS_DS035_COVERAGE;assert.equal(Object.keys(data.states).length,2);for(const [h,rel]of Object.entries(data.assetPaths)){assert.equal(app.assetPaths[h],rel);assert.equal(crypto.createHash('sha256').update(fs.readFileSync(R+'/'+rel)).digest('hex'),h);}});
check('S05 and other original-template suits have no coverage metadata',()=>{for(const id of app.assemblies.dynamicSuitIds)assert.equal(app.assemblies.forSelection(id,'shoe-8','DS035').ds035_s11_coverage,undefined);});
check('Wider suit cannot inherit a coverage layer without exact DS035 request',()=>{for(const id of app.assemblies.widerDS035SuitIds){assert.throws(()=>app.assemblies.forSelection(id,'shoe-8'));assert.throws(()=>app.assemblies.forSelection(id,'shoe-8','DS036'));}});
check('Wear event binds the actual canonical and historical suit identity',()=>{const e=app.createHistoryEvent(s,{id:'coverage_wear_01',localDate:'2026-09-22'});assert.equal(e.canonical_selection.suitId,'S11');assert.equal(e.items.topwear.id,app.aliases.get('S11'));assert.equal(e.items.shirt.id,'shirt-DS035');assert.equal(e.items.tie.id,'T017');});
check('Unchanged source lock retains valid V1.2-format saved sessions',()=>{const db={s:new Map(),getItem(k){return this.s.get(k)??null},setItem(k,v){this.s.set(k,v)}};const store=cx.HEWRSLocalState.create(app,cx.HEWRS_INPUT_SHA256,db);store.saveSession({selection:{...s,suitId:'S05'},origin:'manual',mode:'anchor',localDate:'2026-09-22',context:{occasion:'clinic',requiredFormality:'any'}});const backup=store.exportText();store.restore(store.previewImport(backup));assert.equal(store.snapshot().session.selection.suitId,'S05');assert.equal(store.snapshot().events.length,0);});
